cmake_minimum_required(VERSION 3.4.1)

# Pinball Fantasies for Android -- native build.
#
# One shared library, pbfhost, made of two layers:
#
#   descore/  the shared descore shell (JNI bridge, indexed-color video, audio mixer, ProTracker
#             replayer, touch overlay). Nothing in it knows what game it hosts; it is meant to be
#             carried unchanged between ports.
#   pbf/      Pinball Fantasies itself, translated from the original x86 assembly.
#
# The library name has to match DescoreGameConfig.nativeLibrary (see PbfConfig.java), and the JNI
# symbols in descore_jni.c are named for wootbeer.descore.DescoreView, so neither can be renamed
# without the other.

find_library(log-lib log)
find_library(GLESv1_CM-lib GLESv1_CM)
find_library(EGL-lib EGL)
find_library(android-lib android)

set(DESCORE_SOURCES
		descore/descore_jni.c
		descore/descore_video.c
		descore/descore_audio.c
		descore/audio_mod/descore_mod.c
		descore/touch/descore_touch.c
)

# The game itself: the translated assembly (t1_*.c is generated by tools/gen2.py, t2..t4_*.c by tools/gen_table.py and i1_*.c (intro/chooser) by tools/gen_intro.py) and the
# small DOS/VGA/sound-driver runtime it runs on. musttail (used for the translated jumps) needs clang, which
# is what the NDK ships; -O2 is forced even in debug builds because the translated code relies on tail calls
# being real and is far too slow unoptimized.
set(PBF_SOURCES
		pbf/pbf_main.c
		pbf/pbf_audio.c
		pbf/pbf.c
		pbf/vga.c
		pbf/x86rt.c
		pbf/t1_code.c
		pbf/t1_data.c
		pbf/t1_dispatch.c
		pbf/t1_program.c
		pbf/t2_code.c
		pbf/t2_data.c
		pbf/t2_dispatch.c
		pbf/t2_program.c
		pbf/t3_code.c
		pbf/t3_data.c
		pbf/t3_dispatch.c
		pbf/t3_program.c
		pbf/t4_code.c
		pbf/t4_data.c
		pbf/t4_dispatch.c
		pbf/t4_program.c
		pbf/i1_code.c
		pbf/i1_data.c
		pbf/i1_dispatch.c
		pbf/i1_program.c
)
set_source_files_properties(${PBF_SOURCES} PROPERTIES
		COMPILE_FLAGS "-O2 -Wno-unused-label -Wno-unused-result -Wno-unused-variable")

add_library(pbfhost SHARED ${DESCORE_SOURCES} ${PBF_SOURCES})
target_include_directories(pbfhost PRIVATE pbf)

target_link_libraries(pbfhost
		${log-lib}
		${GLESv1_CM-lib}
		${EGL-lib}
		${android-lib}
		m)
